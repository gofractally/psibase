cmake_minimum_required(VERSION 3.24)
project(psibase_storage_contract LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(../../../../../cmake/psitri.cmake)
psibase_add_psitri()
enable_testing()
add_executable(engine_dependency_smoke engine_dependency_smoke.cpp)
target_link_libraries(engine_dependency_smoke PRIVATE psibase::storage_engine)
add_test(NAME engine_dependency_smoke COMMAND engine_dependency_smoke "${CMAKE_CURRENT_BINARY_DIR}/smoke-database")
find_package(Threads REQUIRED)
add_library(psibase_cpu_clock STATIC ../../src/Watchdog.cpp)
target_include_directories(psibase_cpu_clock PUBLIC ../../include)
target_link_libraries(psibase_cpu_clock PUBLIC Threads::Threads)
add_executable(cpu_accounting_contract cpu_accounting_contract.cpp)
target_link_libraries(cpu_accounting_contract PRIVATE psibase_cpu_clock)
add_test(NAME cpu_accounting_contract COMMAND cpu_accounting_contract)
if(APPLE)
  add_executable(darwin_cpu_contract darwin_cpu_contract.cpp)
  target_link_libraries(darwin_cpu_contract PRIVATE psibase_cpu_clock)
  add_test(NAME darwin_cpu_lifetime_io COMMAND darwin_cpu_contract)
  foreach(mode previous blocked replaced)
    add_test(NAME darwin_cpu_${mode} COMMAND darwin_cpu_contract ${mode})
  endforeach()
  set_tests_properties(darwin_cpu_lifetime_io darwin_cpu_previous darwin_cpu_blocked
                       darwin_cpu_replaced PROPERTIES TIMEOUT 30)
endif()
