cmake_minimum_required(VERSION 3.24)
project(psibase_storage_contract LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(../../../../../cmake/psitri.cmake)
psibase_add_psitri()
enable_testing()
add_executable(engine_dependency_smoke engine_dependency_smoke.cpp)
target_link_libraries(engine_dependency_smoke PRIVATE psibase::storage_engine)
add_test(NAME engine_dependency_smoke COMMAND engine_dependency_smoke "${CMAKE_CURRENT_BINARY_DIR}/smoke-database")
add_library(psibase_storage STATIC ../../src/psitriStorage.cpp)
target_include_directories(psibase_storage PUBLIC ../../include)
target_link_libraries(psibase_storage PRIVATE psibase::storage_engine)
add_executable(storage_contract storage_contract.cpp)
target_link_libraries(storage_contract PRIVATE psibase_storage)
add_test(NAME storage_contract COMMAND storage_contract "${CMAKE_CURRENT_BINARY_DIR}/contract-databases")
add_executable(serialized_contract serialized_contract.cpp)
target_link_libraries(serialized_contract PRIVATE psibase_storage psibase_cpu_clock)
add_test(NAME serialized_contract COMMAND serialized_contract "${CMAKE_CURRENT_BINARY_DIR}/serialized-databases")
set_tests_properties(serialized_contract PROPERTIES TIMEOUT 45)
foreach(name HEART_PAGE_BYTES HEART_PINNED_MB HEART_READ_CACHE HEART_READ_CACHE_SEC)
  add_test(NAME storage_reject_${name} COMMAND ${CMAKE_COMMAND} -E env "${name}=1"
           $<TARGET_FILE:storage_contract> --reject-engine-environment)
endforeach()
find_package(Threads REQUIRED)
add_library(psibase_cpu_clock STATIC ../../src/Watchdog.cpp)
target_include_directories(psibase_cpu_clock PUBLIC ../../include)
target_link_libraries(psibase_cpu_clock PUBLIC Threads::Threads)
add_executable(cpu_accounting_contract cpu_accounting_contract.cpp)
target_link_libraries(cpu_accounting_contract PRIVATE psibase_cpu_clock)
add_test(NAME cpu_accounting_contract COMMAND cpu_accounting_contract)
if(APPLE)
  add_executable(darwin_cpu_contract darwin_cpu_contract.cpp)
  target_link_libraries(darwin_cpu_contract PRIVATE psibase_cpu_clock)
  add_test(NAME darwin_cpu_lifetime_io COMMAND darwin_cpu_contract)
  foreach(mode previous blocked replaced)
    add_test(NAME darwin_cpu_${mode} COMMAND darwin_cpu_contract ${mode})
  endforeach()
  set_tests_properties(darwin_cpu_lifetime_io darwin_cpu_previous darwin_cpu_blocked
                       darwin_cpu_replaced PROPERTIES TIMEOUT 30)
endif()
add_executable(storage_benchmark EXCLUDE_FROM_ALL storage_benchmark.cpp)
target_link_libraries(storage_benchmark PRIVATE psibase_storage psibase_cpu_clock psibase::storage_engine)
option(PSIBASE_STORAGE_BENCHMARK_TRIEDENT "Compare the preserved triedent implementation in the same fixture" OFF)
if(PSIBASE_STORAGE_BENCHMARK_TRIEDENT)
  set(reference_triedent "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../libraries/triedent")
  add_library(psibase_reference_triedent STATIC
      "${reference_triedent}/src/database.cpp" "${reference_triedent}/src/mapping.cpp"
      "${reference_triedent}/src/gc_queue.cpp" "${reference_triedent}/src/ring_allocator.cpp"
      "${reference_triedent}/src/region_allocator.cpp" "${reference_triedent}/src/cache_allocator.cpp")
  target_include_directories(psibase_reference_triedent PUBLIC "${reference_triedent}/include")
  target_link_libraries(psibase_reference_triedent PUBLIC Threads::Threads)
  target_compile_definitions(storage_benchmark PRIVATE PSIBASE_COMPARE_TRIEDENT=1)
  target_link_libraries(storage_benchmark PRIVATE psibase_reference_triedent)
endif()
