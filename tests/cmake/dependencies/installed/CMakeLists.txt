cmake_minimum_required(VERSION 3.20)
project(psibase_installed_psio_proof LANGUAGES CXX)
enable_testing()
find_package(psio CONFIG REQUIRED)
get_target_property(is_imported psio::psio IMPORTED)
if(NOT is_imported)
    message(FATAL_ERROR "Expected the installed PSIO imported target")
endif()
get_target_property(installed_includes psio::psio INTERFACE_INCLUDE_DIRECTORIES)
message(STATUS "Installed PSIO includes: ${installed_includes}")
add_executable(installed-psio-consumer ../consumer.cpp)
target_link_libraries(installed-psio-consumer PRIVATE psio::psio)
add_test(NAME installed-psio-consumer COMMAND installed-psio-consumer)
