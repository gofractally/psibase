cmake_minimum_required(VERSION 3.20)
project(psibase_shared_ucc_proof LANGUAGES CXX)
enable_testing()
set(PSIBASE_UCC_PROOF_MODE consumer CACHE STRING "consumer, collision, or changed-includes")
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/ucc.cmake")
if(PSIBASE_UCC_PROOF_MODE STREQUAL "collision")
    add_library(ucc INTERFACE)
    add_library(ucc::ucc ALIAS ucc)
endif()
psibase_add_ucc()
if(PSIBASE_UCC_PROOF_MODE STREQUAL "changed-includes")
    target_include_directories(ucc INTERFACE "${CMAKE_CURRENT_BINARY_DIR}/unqualified")
endif()
# Model independently activated storage and VM consumers. A second activation
# must reuse the same controlled target and recheck its source/include identity.
psibase_add_ucc()
psibase_acquire_dependency(psizam)
if(TARGET psizam OR TARGET psio OR TARGET psio::psio)
    message(FATAL_ERROR "The source-only native engine evaluated its facade")
endif()
add_executable(shared-ucc-consumer first.cpp second.cpp)
target_include_directories(shared-ucc-consumer PRIVATE "${psizam_SOURCE_DIR}/include")
target_link_libraries(shared-ucc-consumer PRIVATE ucc::ucc)
add_test(NAME shared-ucc-consumer COMMAND shared-ucc-consumer)
